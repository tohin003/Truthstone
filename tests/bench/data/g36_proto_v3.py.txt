"""Prototype G36 v3 (measurement only, never product code): the round-2 settlement's G36 rule.

An existing TOP-LEVEL import statement of a protected test file that only GAINED bindings is an
addition (``protected_extended``), not a weakening, when:
  * its module is unchanged (TS/JS module specifier; Python module and relative level; a plain
    ``import a`` only gains other ``import`` names);
  * every old binding is kept with its alias (TS/JS: and its default, namespace and ``type``
    modifiers);
  * no gained local name is USED in the pristine file.

"Used" (SETTLEMENT C4/G36; t2-2-12, struct-2-8, rd3-2-5/6) is computed by ONE shared helper,
``used_names``, which the product prototype (``ixret_proto_v3.py``) imports from here (parity):
  * Python: every ``ast.Name`` id plus every name an import binds (its alias, or the bound name;
    ``import a.b`` binds ``a``); NEVER attribute names. None when the file does not parse.
  * TS/JS: identifiers in code with comments removed and string literals blanked (template-literal
    text blanked, its ``${...}`` expressions kept as code), except an identifier directly after
    ``.`` or ``?.`` and except an object-literal key (an identifier followed by ``:`` whose previous
    code character is ``{`` or ``,``). Other files: None.

TS/JS imports are compared per STATEMENT (lines from ``import`` to its ``from "<module>"`` joined and
normalised: whitespace, a trailing comma, a final ``;`` and a trailing ``//`` comment do not
matter, nor does the quote style), paired by module when each side has exactly one parsed
statement for it. An extended pair's lines are removed from BOTH texts before ``_script`` runs, so
its test blocks and depths are computed on aligned texts (rd3-2-19). A statement that does not
parse is never paired (stays weakened); CommonJS ``require`` destructuring is out of scope.

Python: before ``old_top - unmatched``, top-level ``ImportFrom`` statements are paired by
(module, level) when each side has exactly one unmatched statement for it, and ``Import``
statements whose old (name, alias) pairs are all kept; a nested import is unchanged. Both paired
statements leave the comparison and ``import extended: <names>`` is reported (rd3-2-20).

install(testedits) switches the grader's classifier to the G36 rule (the ``weakening-g36`` policy).
"""
from __future__ import annotations

import ast
import builtins
import keyword
import re
from collections import Counter
from dataclasses import dataclass

_PY_EXT = (".py", ".pyi")
_JS_EXT = (".ts", ".tsx", ".mts", ".cts", ".js", ".jsx", ".mjs", ".cjs")

# -- a TS/JS code view: comments removed, string literals blanked, newlines kept ----------------------

_REGEX_PREV = set("(,=:[!&|?{};+-*%<>~^")
_REGEX_KEYWORDS = {"return", "typeof", "instanceof", "in", "of", "new", "delete", "void", "throw", "case", "do", "else", "yield", "await"}


def js_code(text: str) -> str:
    """``text`` with comments removed (newlines kept), string and regex literals blanked to ``""``
    / ``//`` shells, template literal text blanked with its ``${...}`` expressions kept as code.
    A best-effort scanner (no full tokenizer): it never raises."""
    out: list[str] = []
    i, n = 0, len(text)
    template_depth: list[int] = []  # brace depth at which each open ${ ... } returns to a template
    brace = 0
    last_sig = ""  # the last significant (non-space) code character emitted
    last_word = ""
    try:
        while i < n:
            c = text[i]
            nxt = text[i + 1] if i + 1 < n else ""
            if c == "/" and nxt == "/":
                j = text.find("\n", i)
                i = n if j < 0 else j
                continue
            if c == "/" and nxt == "*":
                j = text.find("*/", i + 2)
                block = text[i : (n if j < 0 else j + 2)]
                out.append("\n" * block.count("\n"))
                i = n if j < 0 else j + 2
                continue
            if c in ("'", '"'):
                j = i + 1
                while j < n and text[j] != c and text[j] != "\n":
                    j += 2 if text[j] == "\\" else 1
                out.append(c + c)
                i = j + 1 if j < n and text[j] == c else j
                last_sig, last_word = c, ""
                continue
            if c == "`" or (c == "}" and template_depth and template_depth[-1] == brace):
                if c == "}":
                    template_depth.pop()
                # inside template text until ` or ${
                j = i + 1
                newlines = 0
                while j < n:
                    if text[j] == "\\":
                        j += 2
                        continue
                    if text[j] == "`":
                        break
                    if text[j] == "$" and j + 1 < n and text[j + 1] == "{":
                        break
                    if text[j] == "\n":
                        newlines += 1
                    j += 1
                out.append("`" + "\n" * newlines)
                if j < n and text[j] == "`":
                    out.append("`")
                    i = j + 1
                    last_sig, last_word = "`", ""
                elif j < n:  # ${
                    out.append(" ")
                    template_depth.append(brace)
                    i = j + 2
                    last_sig, last_word = "(", ""
                else:
                    i = n
                continue
            if c == "/" and (last_sig in _REGEX_PREV or last_sig == "" or last_word in _REGEX_KEYWORDS):
                # a regex literal: to the closing unescaped / outside a class, on this line
                j = i + 1
                in_class = False
                while j < n and text[j] != "\n":
                    ch = text[j]
                    if ch == "\\":
                        j += 2
                        continue
                    if ch == "[":
                        in_class = True
                    elif ch == "]":
                        in_class = False
                    elif ch == "/" and not in_class:
                        break
                    j += 1
                if j < n and text[j] == "/":
                    j += 1
                    while j < n and (text[j].isalnum() or text[j] == "_"):
                        j += 1  # flags
                    out.append("//")
                    i = j
                    last_sig, last_word = "/", ""
                    continue
                # not a regex after all: a division
            if c == "{":
                brace += 1
            elif c == "}":
                brace = max(0, brace - 1)
            out.append(c)
            if not c.isspace():
                if c.isalnum() or c in "_$":
                    # track the current word for keyword checks
                    k = i
                    while k > 0 and (text[k - 1].isalnum() or text[k - 1] in "_$"):
                        k -= 1
                    last_word = text[k : i + 1]
                    last_sig = "a"
                else:
                    last_sig, last_word = c, ""
            i += 1
    except Exception:  # noqa: BLE001 - a view, never a failure
        return re.sub(r"/\*.*?\*/|//[^\n]*", "", text, flags=re.S)
    return "".join(out)


_IDENT_RE = re.compile(r"(?<![\w$])[A-Za-z_$][\w$]*")


def used_names(path: str, text: str | None) -> frozenset[str] | None:
    """The names a file USES (see the module docstring); None when unknown."""
    if text is None:
        return None
    lower = path.lower()
    if lower.endswith(_PY_EXT):
        try:
            tree = ast.parse(text)
        except (SyntaxError, ValueError, RecursionError, MemoryError):
            return None
        out: set[str] = set()
        for node in ast.walk(tree):
            if isinstance(node, ast.Name):
                out.add(node.id)
            elif isinstance(node, (ast.Import, ast.ImportFrom)):
                for alias in node.names:
                    if alias.name == "*":
                        continue
                    out.add(alias.asname or alias.name.split(".")[0])
        return frozenset(out)
    if lower.endswith(_JS_EXT):
        code = js_code(text)
        out = set()
        for m in _IDENT_RE.finditer(code):
            j = m.start() - 1
            while j >= 0 and code[j] in " \t\r\n":
                j -= 1
            prev = code[j] if j >= 0 else ""
            if prev == "." and not (j >= 2 and code[j - 2 : j + 1] == "..."):
                continue  # a member access: x.Name, x?.Name (a spread ...Name is a use)
            k = m.end()
            while k < len(code) and code[k] in " \t\r\n":
                k += 1
            if k < len(code) and code[k] == ":" and prev in ("{", ",") and code[k + 1 : k + 2] != ":":
                continue  # an object-literal key
            out.add(m.group(0))
        return frozenset(out)
    return None


# -- TS/JS import statements ------------------------------------------------------------------------


@dataclass(frozen=True)
class TsImport:
    module: str
    type_only: bool
    default: str | None
    namespace: str | None
    named: tuple[tuple[str, str, bool], ...]  # (name, local alias, per-name ``type`` flag)


_TS_IMPORT = re.compile(
    r"""^import\s+(?P<type>type\s+(?=[\w${*]))?"""
    r"""(?:(?P<default>[\w$]+)\s*(?:,\s*)?)?"""
    r"""(?:\*\s*as\s+(?P<ns>[\w$]+)\s*)?"""
    r"""(?:\{(?P<names>[^{}]*)\}\s*)?"""
    r"""from\s*(?P<q>['"])(?P<mod>[^'"]+)(?P=q)$"""
)
_TS_NAMED = re.compile(r"^(?P<type>type\s+)?(?P<name>[\w$]+)(?:\s+as\s+(?P<alias>[\w$]+))?$")


def _strip_line_comment(line: str) -> str:
    quote = ""
    for k, ch in enumerate(line):
        if quote:
            if ch == quote:
                quote = ""
            continue
        if ch in ("'", '"', "`"):
            quote = ch
        elif ch == "/" and line[k + 1 : k + 2] == "/":
            return line[:k]
    return line


def parse_ts_import(statement: str) -> TsImport | None:
    text = re.sub(r"\s+", " ", statement).strip()
    text = re.sub(r"\s*;\s*$", "", text)
    m = _TS_IMPORT.match(text)
    if not m:
        return None
    if m["default"] == "type" and not m["type"]:
        return None  # ambiguous: leave it weakened
    named: list[tuple[str, str, bool]] = []
    if m["names"] is not None:
        for item in m["names"].split(","):
            item = item.strip()
            if not item:
                continue
            nm = _TS_NAMED.match(item)
            if not nm:
                return None
            named.append((nm["name"], nm["alias"] or nm["name"], bool(nm["type"])))
    if m["default"] is None and m["ns"] is None and m["names"] is None:
        return None
    return TsImport(m["mod"], bool(m["type"]), m["default"], m["ns"], tuple(named))


def ts_imports(text: str) -> list[tuple[int, int, TsImport | None, str]]:
    """Top-level import statements: (first line, last line, parsed or None, normalised text)."""
    lines = text.splitlines()
    out: list[tuple[int, int, TsImport | None, str]] = []
    i = 0
    while i < len(lines):
        line = lines[i]
        if re.match(r"import\b(?!\s*\()", line) and not re.match(r"import\s*['\"]", line):
            parts: list[str] = []
            end = None
            for j in range(i, min(len(lines), i + 60)):
                parts.append(_strip_line_comment(lines[j]))
                joined = " ".join(parts)
                if re.search(r"""\bfrom\s*['"][^'"]+['"]\s*;?\s*$""", joined):
                    end = j
                    break
            if end is None:
                out.append((i, i, None, line.strip()))
                i += 1
                continue
            joined = " ".join(parts)
            out.append((i, end, parse_ts_import(joined), re.sub(r"\s+", " ", joined).strip()))
            i = end + 1
            continue
        i += 1
    return out


def ts_import_gained(old: TsImport, new: TsImport, used: frozenset[str]) -> list[str] | None:
    """The gained local names when ``new`` only extends ``old`` and none is used, else None."""
    if (old.module, old.type_only, old.default, old.namespace) != (new.module, new.type_only, new.default, new.namespace):
        return None
    old_set, new_set = set(old.named), set(new.named)
    if not old_set < new_set:
        return None
    gained = sorted({alias for _, alias, _ in new_set - old_set})
    if not gained or any(name in used for name in gained):
        return None
    return gained


def ts_extended_pairs(before: str, after: str) -> list[tuple[tuple[int, int], tuple[int, int], list[str]]]:
    used = used_names("x.ts", before) or frozenset()
    old_stmts, new_stmts = ts_imports(before), ts_imports(after)
    by_mod_old: dict[str, list] = {}
    by_mod_new: dict[str, list] = {}
    for first, last, parsed, norm in old_stmts:
        if parsed is not None:
            by_mod_old.setdefault(parsed.module, []).append((first, last, parsed, norm))
    for first, last, parsed, norm in new_stmts:
        if parsed is not None:
            by_mod_new.setdefault(parsed.module, []).append((first, last, parsed, norm))
    pairs = []
    for mod, olds in by_mod_old.items():
        news = by_mod_new.get(mod, [])
        if len(olds) != 1 or len(news) != 1:
            continue  # pairing needs exactly one statement for the module on each side (comp-2-30)
        (of, ol, op, _), (nf, nl, np_, _) = olds[0], news[0]
        if op == np_:
            continue  # the same statement: nothing to pair
        gained = ts_import_gained(op, np_, used)
        if gained:
            pairs.append(((of, ol), (nf, nl), gained))
    return pairs


def _drop_lines(text: str, ranges: list[tuple[int, int]]) -> str:
    lines = text.splitlines()
    drop = {k for first, last in ranges for k in range(first, last + 1)}
    return "\n".join(line for k, line in enumerate(lines) if k not in drop) + ("\n" if text.endswith("\n") else "")


# -- Python import statements ------------------------------------------------------------------------


def _py_pairs(alias_list) -> set[tuple[str, str | None]]:
    return {(a.name, a.asname) for a in alias_list}


def _py_bound(pairs) -> set[str]:
    return {asname or name.split(".")[0] for name, asname in pairs}


def py_extended_pairs(testedits, old_top: list, new_top: list, used: frozenset[str]):
    """(old statement, new statement, gained names) for top-level imports that only gained unused
    bindings; statements identical on both sides are matched first and never paired."""
    old_d = Counter(testedits._dump(s) for s in old_top)
    new_d = Counter(testedits._dump(s) for s in new_top)
    common = old_d & new_d
    left_old, left_new = [], []
    seen = Counter()
    for s in old_top:
        d = testedits._dump(s)
        if seen[d] < common[d]:
            seen[d] += 1
            continue
        left_old.append(s)
    seen = Counter()
    for s in new_top:
        d = testedits._dump(s)
        if seen[d] < common[d]:
            seen[d] += 1
            continue
        left_new.append(s)
    pairs = []
    used_new: set[int] = set()
    for s in left_old:
        if isinstance(s, ast.ImportFrom):
            key = (s.module, s.level)
            olds = [o for o in left_old if isinstance(o, ast.ImportFrom) and (o.module, o.level) == key]
            news = [k for k, o in enumerate(left_new) if isinstance(o, ast.ImportFrom) and (o.module, o.level) == key]
            if len(olds) != 1 or len(news) != 1 or news[0] in used_new:
                continue
            twin = left_new[news[0]]
        elif isinstance(s, ast.Import):
            old_pairs = _py_pairs(s.names)
            news = [k for k, o in enumerate(left_new) if isinstance(o, ast.Import) and k not in used_new and old_pairs < _py_pairs(o.names)]
            if len(news) != 1:
                continue
            twin = left_new[news[0]]
        else:
            continue
        old_pairs, new_pairs = _py_pairs(s.names), _py_pairs(twin.names)
        if not old_pairs < new_pairs or any(name == "*" for name, _ in new_pairs):
            continue
        gained = sorted(_py_bound(new_pairs - old_pairs))
        if not gained or any(name in used for name in gained):
            continue
        used_new.add(news[0])
        pairs.append((s, twin, gained))
    return pairs


def install(testedits) -> None:
    """Switch ``testedits`` (the bench grader's classifier) to the G36 rule."""
    orig_script = testedits._script

    def _script(before: str, after: str):
        try:
            pairs = ts_extended_pairs(before, after)
        except Exception:  # noqa: BLE001
            pairs = []
        if not pairs:
            return orig_script(before, after)
        b2 = _drop_lines(before, [p[0] for p in pairs])
        a2 = _drop_lines(after, [p[1] for p in pairs])
        weakened, extended = orig_script(b2, a2)
        extended = [*extended, *(f"import extended: {', '.join(g)}" for _, _, g in pairs)]
        return weakened, extended

    testedits._script = _script

    def _python(before: str, after: str, existing: bool):
        try:
            old = ast.parse(before)
        except SyntaxError:
            return ["changed a file that did not parse"], []
        try:
            new = ast.parse(after)
        except SyntaxError:
            return ["no longer parses"], []
        weakened: list[str] = []
        old_units, new_units = testedits._units(old), testedits._units(new)
        for name, dump in old_units.items():
            if name not in new_units:
                weakened.append(f"{name.removeprefix('class:')} removed")
            elif new_units[name] != dump:
                weakened.append(f"{name.removeprefix('class:')} changed")
        old_stmts = [s for s in testedits._strip_docstring(old.body) if not testedits._is_def(s)]
        new_top = [s for s in testedits._strip_docstring(new.body) if not testedits._is_def(s)]
        extended: list[str] = []
        if existing:
            used = used_names("x.py", before) or frozenset()
            for o, t, gained in py_extended_pairs(testedits, old_stmts, new_top, used):
                old_stmts = [s for s in old_stmts if s is not o]
                new_top = [s for s in new_top if s is not t]
                extended.append(f"import extended: {', '.join(gained)}")
        old_top = Counter(testedits._dump(s) for s in old_stmts)
        unmatched = Counter(testedits._dump(s) for s in new_top)
        if old_top - unmatched:
            weakened.append("module-level code changed")
        for node in new_top:
            dump = testedits._dump(node)
            if old_top[dump]:
                old_top[dump] -= 1
                continue
            why = testedits._new_top_level(node)
            if why:
                weakened.append(why)
        for name in sorted(new_units.keys() - old_units.keys()):
            why = testedits._new_unit(name, new) if existing else ""
            if why:
                weakened.append(why)
            elif not name.startswith("class:"):
                extended.append(f"{name} added")
        return weakened, extended

    testedits._python = _python
